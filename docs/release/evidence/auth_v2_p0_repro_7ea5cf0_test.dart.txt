    addTearDown(container.dispose);
    await container.read(authControllerProvider.notifier).completeBootstrap();
    final router = container.read(appRouterProvider);
    addTearDown(router.dispose);
    await tester.pumpWidget(
      UncontrolledProviderScope(
        container: container,
        child: MaterialApp.router(
          routerConfig: router,
          locale: const Locale('fr'),
          supportedLocales: AppLocalizations.supportedLocales,
          localizationsDelegates: const [
            AppLocalizations.delegate,
            GlobalMaterialLocalizations.delegate,
            GlobalWidgetsLocalizations.delegate,
            GlobalCupertinoLocalizations.delegate,

      ),

    await tester.pump(const Duration(milliseconds: 600));

    await tester.tap(find.byKey(const ValueKey('gateway-google-auth')));
    for (var i = 0; i < 30; i++) {
      await tester.pump(const Duration(milliseconds: 50));
    }

    expect(
      container.read(authControllerProvider).status,
      AuthStatus.authenticated,
    );
    expect(router.state.uri.path, AppRoutes.parentHome);
  });

  test(

class _Repository implements AuthRepository, AuthSessionResolver {
  String? signedInUid;

  @override
  Future<AuthSessionResolution> resolveCurrentSession() async {
    if (signedInUid == null) {
      return const AuthSessionResolution(
        kind: AuthSessionResolutionKind.unauthenticated,
      );
    }
    return AuthSessionResolution(
      kind: AuthSessionResolutionKind.authenticated,
      firebaseUid: signedInUid,
      user: AuthUserData(
        uid: signedInUid!,
        email: '',
        role: AppRole.parent,
        firstName: 'Claire',
        lastName: '',
        profileCompleted: true,
      ),
    );
  }

  @override
  dynamic noSuchMethod(Invocation invocation) => super.noSuchMethod(invocation);
}

class _SignsInExistingParent implements GoogleAuthGateway {
  _SignsInExistingParent(this.repository);
  final _Repository repository;

  @override
  Future<GoogleAuthResult> signIn() async {
    repository.signedInUid = 'parent-uid';
    return const GoogleAuthSuccess(
      uid: 'parent-uid',
      roles: [AppRole.parent],
      profileCompleted: true,
    );
  }

  @override
  Future<AccountLinkingState> linkGoogleCredential(
    User currentUser,
    AuthCredential credential,
  ) => throw UnimplementedError();

  @override
  Future<void> signOut() async {}
}
